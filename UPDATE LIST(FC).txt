# RHYTHM N DRIVE - Enhanced Edition
Full implementation plan untuk menambah features baru tanpa mengubah logik sedia ada

## Ringkasan Projek
Web-based racing game dengan rhythm elements. Menambah 15+ kereta baru (termasuk kereta tempatan Malaysia), tutorial system, visual feedback, sound effects, loading indicators, weather system, achievement system, AI opponent, customization, difficulty settings, dan season pass.

```javascript
    drop.style.animationDelay = Math.random() * 2 + 's';

    rainContainer.appendChild(drop);
  }
  document.body.appendChild(rainContainer);
```

**5. AI Opponent System:**
```javascript
// AI opponent configuration
let aiEnabled = false;
let aiDifficulty = 'medium'; // easy, medium, hard
let aiX = 1; // AI lane position
let aiY = -200; // AI vertical position
let aiSpeed = 2;
let aiLives = 3;
let aiCash = 0;

const AI_CONFIG = {
  easy: { reactionTime: 800, mistakeChance: 0.3, speedMult: 0.8 },
  medium: { reactionTime: 500, mistakeChance: 0.15, speedMult: 1 },
  hard: { reactionTime: 200, mistakeChance: 0.05, speedMult: 1.2 }
};

function updateAI(dt) {
  if (!aiEnabled) return;
  
  const config = AI_CONFIG[aiDifficulty];
  
  // AI movement logic
  aiY += aiSpeed * config.speedMult * getDifficultyMultiplier() * dt * 60;
  
  // AI lane switching (simple obstacle avoidance)
  if (Math.random() < 0.02 && Math.random() > config.mistakeChance) {
    const obstacles = document.querySelectorAll('.cell[data-content="💥"]');
    let shouldMove = false;
    let targetLane = aiX;
    
    obstacles.forEach(obs => {
      const obsLane = parseInt(obs.dataset.x);
      const obsRow = parseInt(obs.dataset.y);
      if (obsLane === aiX && obsRow < 3) {
        shouldMove = true;
        targetLane = aiX === 0 ? 1 : aiX === 2 ? 1 : (Math.random() > 0.5 ? 0 : 2);
      }
    });
    
    if (shouldMove) {
      setTimeout(() => { aiX = targetLane; }, config.reactionTime);
    }
  }
  
  // Random movement (mistakes)
  if (Math.random() < config.mistakeChance * 0.1) {
    aiX = Math.max(0, Math.min(2, aiX + (Math.random() > 0.5 ? 1 : -1)));
  }
  
  // Check AI collisions with items
  checkAICollisions();
}

function checkAICollisions() {
  const cells = document.querySelectorAll('.cell');
  cells.forEach(cell => {
    const cx = parseInt(cell.dataset.x);
    const cy = parseInt(cell.dataset.y);
    const content = cell.dataset.content;
    
    if (cx === aiX && cy === 2 && content) {
      if (content === '💥') {
        aiLives--;
        playSound('collision');
        if (aiLives <= 0) {
          // AI game over
          aiEnabled = false;
        }
      } else if (content === '💰') {
        aiCash += 10;
        cell.dataset.content = '';
        cell.textContent = '';
      } else if (content === '💎') {
        // AI collects gem
        cell.dataset.content = '';
        cell.textContent = '';
      }
    }
  });
}

function renderAI() {
  if (!aiEnabled) return;
  
  // Render AI car on board
  const cells = document.querySelectorAll('.cell');
  cells.forEach(cell => {
    const cx = parseInt(cell.dataset.x);
    const cy = parseInt(cell.dataset.y);
    
    // Remove old AI car
    if (cell.querySelector('.ai-car')) {
      cell.querySelector('.ai-car').remove();
    }
    
    // Draw AI car at its position
    if (cx === aiX && cy === 2) {
      const aiCar = document.createElement('div');
      aiCar.className = 'ai-car';
      aiCar.innerHTML = getCarHTML('saga', 'ai-opponent');
      cell.appendChild(aiCar);
    }
  });
  
  // Update AI UI
  document.getElementById('ai-lives').textContent = aiLives;
  document.getElementById('ai-cash').textContent = aiCash;
}
```

**8. Season Pass System:**
```javascript
// Season Pass configuration
let seasonPassData = {
  tier: 0,
  xp: 0,
  isPremium: false,
  startDate: Date.now(),
  claimed: []
};

const SEASON_TIERS = [];
for (let i = 0; i < 50; i++) {
  SEASON_TIERS.push({
    tier: i + 1,
    xpRequired: 1000 + (i * 200),
    freeReward: i % 5 === 0 ? { type: 'diamond', amount: 5 } : { type: 'money', amount: 500 + (i * 100) },
    premiumReward: i % 10 === 0 ? { type: 'car', carKey: 'exclusive_' + i } : { type: 'customization', item: 'exclusive_decal_' + i }
  });
}

function addSeasonXP(amount) {
  seasonPassData.xp += amount;
  
  // Check tier up
  const currentTier = seasonPassData.tier;
  const nextTierXP = SEASON_TIERS[currentTier]?.xpRequired || 99999;
  
  if (seasonPassData.xp >= nextTierXP && currentTier < 50) {
    seasonPassData.tier++;
    seasonPassData.xp -= nextTierXP;
    showSeasonTierUp(seasonPassData.tier);
    checkAchievement('season_starter'); // If tier 10
    if (seasonPassData.tier === 50) checkAchievement('season_master');
  }
  
  saveSeasonPassData();
}

function showSeasonTierUp(tier) {
  const popup = document.createElement('div');
  popup.className = 'season-tierup-popup';
  popup.innerHTML = `
    <h2>🎉 SEASON TIER UP! 🎉</h2>
    <p class="tier-number">TIER ${tier}</p>
    <p>Check your rewards in Season Pass!</p>
  `;
  document.body.appendChild(popup);
  playSound('levelup');
  
  setTimeout(() => popup.remove(), 3000);
}

function claimSeasonReward(tier, isPremium) {
  const reward = SEASON_TIERS[tier - 1];
  if (!reward) return;
  
  const rewardData = isPremium ? reward.premiumReward : reward.freeReward;
  
  if (rewardData.type === 'money') {
    userData.money += rewardData.amount;
  } else if (rewardData.type === 'diamond') {
    userData.diamonds += rewardData.amount;
  } else if (rewardData.type === 'car') {
    userData.ownedCars.push(rewardData.carKey);
  } else if (rewardData.type === 'customization') {
    // Unlock customization item
    if (!userData.unlockedCustomizations) userData.unlockedCustomizations = [];
    userData.unlockedCustomizations.push(rewardData.item);
  }
  
  seasonPassData.claimed.push(`${tier}-${isPremium ? 'premium' : 'free'}`);
  saveUserData();
  saveSeasonPassData();
}
```

```javascript
  const box = document.getElementById('tutorial-box');
  
  overlay.classList.remove('hidden');
  
  box.innerHTML = `
    <h2>${step.title}</h2>
    <p>${step.message}</p>
    <div class="tutorial-progress">${tutorialStep + 1} / ${TUTORIAL_STEPS.length}</div>
    <button onclick="nextTutorialStep()" class="menu-btn">
      ${tutorialStep === TUTORIAL_STEPS.length - 1 ? 'Start Racing!' : 'Next'}
    </button>
    <button onclick="skipTutorial()" class="menu-btn btn-danger">Skip</button>
  `;
  
  // Highlight target element
  if (step.target && step.highlight) {
    const target = document.querySelector(step.target);
    if (target) {
      target.classList.add('tutorial-highlight');
    }
  }
}

function nextTutorialStep() {
  // Remove previous highlight
  document.querySelectorAll('.tutorial-highlight').forEach(el => {
    el.classList.remove('tutorial-highlight');
  });
  
  tutorialStep++;
  showTutorialStep();
}

function skipTutorial() {
  endTutorial();
}

function endTutorial() {
  tutorialActive = false;
  document.getElementById('tutorial-overlay').classList.add('hidden');
  document.querySelectorAll('.tutorial-highlight').forEach(el => {
    el.classList.remove('tutorial-highlight');
  });
  
  // Mark tutorial as completed
  userData.tutorialCompleted = true;
  saveUserData();
}
```

```javascript
// Particle system for effects
function createParticles(x, y, color, count = 10) {
  for (let i = 0; i < count; i++) {
    const particle = document.createElement('div');
    particle.className = 'particle';
    particle.style.left = x + 'px';
    particle.style.top = y + 'px';
    particle.style.background = color;
    
    const angle = (Math.PI * 2 * i) / count;
    const velocity = 2 + Math.random() * 3;
    const vx = Math.cos(angle) * velocity;
    const vy = Math.sin(angle) * velocity;
    
    particle.style.setProperty('--vx', vx + 'px');
    particle.style.setProperty('--vy', vy + 'px');
    
    document.body.appendChild(particle);
    
    setTimeout(() => particle.remove(), 1000);
  }
}

function screenShake(intensity = 10, duration = 200) {
  document.body.style.animation = `shake ${duration}ms ease-in-out`;
  setTimeout(() => {
    document.body.style.animation = '';
  }, duration);
}

function createSpeedLines() {
  const container = document.getElementById('speed-lines');
  if (!container) {
    const newContainer = document.createElement('div');
    newContainer.id = 'speed-lines';
    newContainer.className = 'speed-lines-container';
    document.getElementById('game-container').appendChild(newContainer);
  }
  
  for (let i = 0; i < 20; i++) {
    const line = document.createElement('div');
    line.className = 'speed-line';
    line.style.left = Math.random() * 100 + '%';
    line.style.animationDelay = Math.random() * 1 + 's';
    document.getElementById('speed-lines').appendChild(line);
  }
}

function removeSpeedLines() {
  const container = document.getElementById('speed-lines');
  if (container) container.remove();
}
```

## Integration Notes

### Integrating with Existing Code:

1. **Kereta Baru**: Tambah dalam existing `CARS` object, jangan replace
2. **Sound System**: Panggil `playSound()` dalam existing functions:
   - Collision: `enhanceCollisionEffects()` dalam collision check
   - Coin collect: `playSound('coin')` bila collect coin
   - Nitro: `playSound('nitro_on')` bila activate nitro
   
3. **Weather System**: Panggil `setWeather()` sebelum start race
4. **AI System**: Panggil `updateAI(dt)` dan `renderAI()` dalam existing `gameLoop`
5. **Achievement Checks**: Tambah `checkAchievement()` calls dalam:
   - End of race: `checkAchievement('first_race')`
   - Level milestones: `checkAchievement('speed_demon')` bila level 10
   - Collection milestones: Check coin/gem counts
   
6. **Season XP**: Panggil `addSeasonXP()` selepas setiap race
7. **Tutorial**: Panggil `startTutorial()` untuk first-time players
